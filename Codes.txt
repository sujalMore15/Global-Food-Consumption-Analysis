# ============================================================
# 1. CORE KPI MEASURES
# ============================================================

# Total Food Consumption

```DAX
Total Food Consumption =
SUM('Food Consumption'[Consumption])
```

# Average Food Consumption

```DAX

```


# ============================================================
# 10. TIME INTELLIGENCE MEASURES
# ============================================================

# Previous Year Consumption

```DAX
Previous Year Consumption =
CALCULATE(
    [Total Food Consumption],
    DATEADD(
        'Food Consumption'[Year],
        -1,
        YEAR
    )
)
```

# Year-over-Year Change

```DAX
YoY Change =
[Total Food Consumption] -
[Previous Year Consumption]
```

# Year-over-Year Growth %

```DAX
YoY Growth % =
DIVIDE(
    [YoY Change],
    [Previous Year Consumption],
    0
)
```

# 11. TOP COUNTRY / AREA ANALYSIS
# ============================================================

# Rank Area by Consumption

```DAX
Area Consumption Rank =
RANKX(
    ALL('Food Consumption'[Area]),
    [Total Food Consumption],
    ,
    DESC,
    DENSE
)
```

# Rank Country by Consumption

```DAX
Country Consumption Rank =
RANKX(
    ALL('Food Consumption'[Country]),
    [Total Food Consumption],
    ,
    DESC,
    DENSE
)
```
